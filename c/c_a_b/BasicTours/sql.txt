CREATE TABLE IF NOT EXISTS BasicTours (
    tour_code TEXT PRIMARY KEY CHECK(length(tour_code) <= 12),
    tour_category TEXT,
    line_name TEXT,
    series_name TEXT,
    tour_name TEXT NOT NULL CHECK(length(tour_name) <= 30),
    foreign_tour_no TEXT,
    product_name TEXT,
    tour_days INTEGER,
    tour_nights INTEGER,
    tour_area TEXT,
    tour_country TEXT,
    departure_airport TEXT,
    dev_features TEXT,
    itinerary_features TEXT,
    special_arrange TEXT,
    sales_features TEXT,
    ticket_desc TEXT,
    departure_notes TEXT,
    b2b_notes TEXT,
    included_items TEXT,
    excluded_items TEXT
);
